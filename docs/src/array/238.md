## 算法步骤
1. 构建前缀乘积数组
2. 构建后缀乘积数组
3. 针对i, res[i] = left[i] * right[i]

```java
class Solution {
    public int[] productExceptSelf(int[] nums) {
        int len = nums.length;
        int[] left = new int[len];
        int[] right = new int[len];

        left[0] = 1;
        for (int i = 1; i < len; i++) {
            left[i] = nums[i-1] * left[i-1];
        }
        right[len-1] = 1;
        for (int i = len-2; i >= 0; i--) {
            right[i] = nums[i+1] * right[i+1];
        }
        int[] res = new int[len];
        for (int i = 0; i < len; i++) {
            res[i] = left[i] * right[i];
        }
        return res;
    }
}
```
